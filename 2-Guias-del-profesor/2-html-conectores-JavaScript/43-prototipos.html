43 - Prototipos en JavaScript

¿Qué es un Prototipo?

Un prototipo es un objeto que sirve como plantilla para crear otros objetos. JavaScript usa prototipos para herencia.

// Función constructora function Persona(nombre, edad) { this.nombre = nombre; this.edad = edad; } // Agregar método al prototipo Persona.prototype.presentarse = function() { return "Hola, soy " + this.nombre; }; let persona = new Persona("Juan", 30); console.log(persona.presentarse());

function Animal(nombre) { this.nombre = nombre; } Animal.prototype.hablar = function() { return this.nombre + " hace un sonido"; }; function Perro(nombre, raza) { Animal.call(this, nombre); this.raza = raza; } Perro.prototype = Object.create(Animal.prototype); Perro.prototype.constructor = Perro; Perro.prototype.hablar = function() { return this.nombre + " ladra"; };

Ventajas de prototipos

  • Eficiencia de memoria: Los métodos se comparten, no se duplican
  • Herencia: Permite que objetos hereden de otros
  • Flexibilidad: Puedes agregar propiedades a múltiples objetos a la vez
