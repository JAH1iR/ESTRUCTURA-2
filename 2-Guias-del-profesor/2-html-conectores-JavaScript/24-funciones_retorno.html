24 - Funciones con Retorno

¿Qué es return?

Return permite que una función devuelva un valor que puede ser usado después.

function sumar(a, b) { return a + b; // Devuelve el resultado } let resultado = sumar(5, 3); // resultado = 8 console.log(resultado); // Imprime: 8

Sintaxis

function nombreFuncion(parametros) { let resultado = operación; return resultado; // Devuelve el resultado } let variable = nombreFuncion(argumentos); // Recibe el valor

Ejemplo 1: Sumar y retornar

function sumar(a, b) { return a + b; } let suma = sumar(10, 5); console.log(suma); // Salida: 15

Ejemplo 2: Multiplicar y retornar

function multiplicar(a, b) { return a * b; } let producto = multiplicar(4, 7); console.log(producto); // Salida: 28

Ejemplo 3: Calcular promedio

function calcularPromedio(a, b, c) { return (a + b + c) / 3; } let promedio = calcularPromedio(8, 9, 7); console.log(promedio); // Salida: 8

Ejemplo 4: Verificar mayor de edad (return booleano)

function esAdulto(edad) { return edad >= 18; } let resultado1 = esAdulto(25); // true let resultado2 = esAdulto(15); // false console.log(resultado1); // Salida: true console.log(resultado2); // Salida: false

Ejemplo 5: Crear nombre completo

function crearNombreCompleto(nombre, apellido) { return nombre + " " + apellido; } let nombreCompleto = crearNombreCompleto("Juan", "Pérez"); console.log(nombreCompleto); // Salida: Juan Pérez

Diferencia: Console.log vs Return

// Sin return - solo imprime function saludarSinReturn() { console.log("Hola"); } // Con return - devuelve un valor function saludarConReturn() { return "Hola"; } let resultado = saludarConReturn(); // Puedo guardar en variable console.log(resultado); // Hola (puedo usar el valor después)
