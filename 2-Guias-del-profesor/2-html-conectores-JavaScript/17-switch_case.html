17 - Control de flujo: Switch-Case

¿Qué es Switch-Case?

Switch es una alternativa a múltiples if-else cuando comparas una variable con varios valores específicos.

switch (valor) { case valor1: // código si es valor1 break; case valor2: // código si es valor2 break; default: // si no coincide con ninguno }

Ejemplo 1: Día de la semana

let dia = 1; let nombre; switch (dia) { case 1: nombre = "Lunes"; break; case 2: nombre = "Martes"; break; case 3: nombre = "Miércoles"; break; default: nombre = "Día inválido"; } console.log(nombre);

Ejemplo 2: Calificación de películas

let calificacion = 'PG'; switch (calificacion) { case 'G': console.log("Apta para todos los públicos"); break; case 'PG': console.log("Se recomienda supervisión parental"); break; case 'PG-13': console.log("Puede no ser apta para menores de 13"); break; case 'R': console.log("Requiere acompañante mayor de edad"); break; default: console.log("Calificación desconocida"); }

Ejemplo 3: Operación matemática

let operador = '+'; let a = 10; let b = 5; let resultado; switch (operador) { case '+': resultado = a + b; break; case '-': resultado = a - b; break; case '*': resultado = a * b; break; case '/': resultado = a / b; break; default: resultado = "Operador no válido"; } console.log(resultado);


Ejemplo 4: Estación del año

let mes = 3; let estacion; switch (mes) { case 12: case 1: case 2: estacion = "Invierno"; break; case 3: case 4: case 5: estacion = "Primavera"; break; // ... }
