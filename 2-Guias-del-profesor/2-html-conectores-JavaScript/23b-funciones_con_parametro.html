23b - Funciones con Parámetros

¿Qué son los parámetros?

Los parámetros permiten que una función reciba valores para trabajar con ellos.

// Definir función con parámetros function saludar(nombre) { console.log("Hola " + nombre); } // Llamar la función pasando un valor saludar("Juan"); // Imprime: Hola Juan saludar("María"); // Imprime: Hola María

Sintaxis

function nombreFuncion(parametro1, parametro2) { // usar parametro1 y parametro2 } nombreFuncion(valor1, valor2); // Pasar valores

Ejemplo 1: Saludar con nombre

function saludar(nombre) { console.log("Hola " + nombre); } saludar("Carlos"); // Salida: Hola Carlos

Ejemplo 2: Sumar dos números

function sumar(a, b) { console.log(a + b); } sumar(5, 3); // Salida: 8 sumar(10, 20); // Salida: 30

Ejemplo 3: Multiplicar número

function multiplicar(numero, factor) { console.log(numero + " x " + factor + " = " + (numero * factor)); } multiplicar(5, 3); // Salida: 5 x 3 = 15 multiplicar(10, 4); // Salida: 10 x 4 = 40

Ejemplo 4: Presentarse con edad

function presentarse(nombre, edad, profesion) { console.log("Mi nombre es " + nombre); console.log("Tengo " + edad + " años"); console.log("Soy " + profesion); } presentarse("Juan", 30, "Ingeniero"); presentarse("María", 25, "Doctora");



Ejemplo 5: Verificar edad

function esAdulto(edad) { if (edad >= 18) { console.log("Eres mayor de edad"); } else { console.log("Eres menor de edad"); } } esAdulto(25); // Salida: Eres mayor de edad esAdulto(15); // Salida: Eres menor de edad
